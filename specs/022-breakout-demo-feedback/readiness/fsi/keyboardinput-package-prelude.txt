
## dotnet fsi scripts/keyboardinput-package-prelude.fsx
$ dotnet fsi scripts/keyboardinput-package-prelude.fsx
keyboard-package-init-effects=[StateDisplayChanged { PressedKeys = []
                       ActiveLayout = "default"
                       ActiveModeStack = []
                       PendingSequence = []
                       LastCommand = None }]
keyboard-package-command=Some "open"
keyboard-package-key-effects=[KeyStateChanged ["K"]; CommandResolved "open"]
keyboard-package-recovered-pressed=set []
keyboard-package-recovery-effects=[KeyStateChanged []; ModeChanged []; PendingSequenceChanged [];
 ReportKeyboardDiagnostic
   { Code = "FocusLostRecovered"
     Severity = "Warning"
     Message = "Focus loss cleared pressed keys and temporary modes."
     Key = None }]
keyboard-package-display={ PressedKeys = []
  ActiveLayout = "default"
  ActiveModeStack = []
  PendingSequence = []
  LastCommand = Some "open" }

exit-code=0
